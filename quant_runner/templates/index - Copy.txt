    <button onclick="openUploadModal()">New Strategy</button>

    <h2>Leg Status</h2>
    <table id="legsTable"></table>

    <!-- Upload/Edit Modal -->
    <div id="strategyModal">
      <h3 id="modalTitle">Upload Strategy</h3>
      <form id="strategyForm">
        <input name="instance_name" placeholder="Instance Name"><br>
        <input name="symbol" placeholder="Symbol"><br>
        <input name="leg1" placeholder="Leg1"><br>
        <input name="leg2" placeholder="Leg2"><br>
        <input name="straddle_width" type="number" step="0.1"><br>
        <input name="start_time" placeholder="HH:MM:SS"><br>
        <input name="squareoff_time" placeholder="HH:MM:SS"><br>
        <input name="end_time" placeholder="HH:MM:SS"><br>
        <input name="expiry" placeholder="27-MAR-26"><br>
        <button type="submit">Save</button>
        <button type="button" onclick="closeModal()">Cancel</button>
      </form>
    </div>

<script>
let editingId = null;
let tableData = [];

function openUploadModal(){
    editingId = null;
    document.getElementById("modalTitle").innerText = "Upload Strategy";
    document.getElementById("strategyForm").reset();
    document.getElementById("strategyModal").style.display="block";
}

function editStrategyById(id){
    const row = tableData.find(r => r.id === id);
    if(!row) return;
    editingId = id;
    document.getElementById("modalTitle").innerText = "Edit Strategy";
    const form = document.getElementById("strategyForm");
    Object.keys(row).forEach(k=>{
        if(form[k]) form[k].value = row[k] || "";
    });
    document.getElementById("strategyModal").style.display="block";
}

function closeModal(){
    document.getElementById("strategyModal").style.display="none";
}

document.getElementById("strategyForm").onsubmit = async function(e){
    e.preventDefault();
    const formData = new FormData(e.target);
    const jsonData = {};
    formData.forEach((v,k)=>jsonData[k]=v);

    let url = "/upload_strategy";
    let method = "POST";
    if(editingId){   // editing existing strategy
        url = "/edit_strategy/"+editingId;
        method = "PUT";
    }

    const res = await fetch(url,{
        method,
        headers:{"Content-Type":"application/json"},
        body: JSON.stringify(jsonData)
    });
    const data = await res.json();
    alert(data.message);
    closeModal();
    loadTables();
};

async function renderTable(endpoint, elementId){
    const res = await fetch(endpoint);
    const data = await res.json();
    if(data.length === 0){
        document.getElementById(elementId).innerHTML = "<tr><td>No data found</td></tr>";
        return;
    }
    if(endpoint==="/view_strategies") tableData = data;

    let html = "<tr>";
    Object.keys(data[0]).forEach(k => html += `<th>${k}</th>`);
    if(endpoint==="/view_strategies") html += "<th>Enabled</th><th>Actions</th>";
    html += "</tr>";

    data.forEach(r=>{
        html += "<tr>";
        Object.keys(r).forEach(k => {
            let val = r[k] === null ? "" : r[k];
            html += `<td>${val}</td>`;
        });
        if(endpoint==="/view_strategies"){
            const checked = r.enabled==1 ? "checked" : "";
            html += `<td><input type="checkbox" ${checked} onchange="toggleStrategy(${r.id},this.checked)"></td>`;
            html += `<td><button onclick='editStrategyById(${r.id})'>Edit</button></td>`;
        }
        html += "</tr>";
    });
    document.getElementById(elementId).innerHTML = html;
}

function loadTables(){
    renderTable("/view_strategies","strategiesTable");
    renderTable("/view_leg_status","legsTable");
}

async function toggleStrategy(id,enabled){
    const res = await fetch("/toggle_strategy/"+id,{
        method:"PUT",
        headers:{"Content-Type":"application/json"},
        body: JSON.stringify({enabled: enabled?1:0})
    });
    const data = await res.json();
    alert(data.message);
}


function loadTables(){
    renderTable("/view_strategies","strategiesTable");
    renderTable("/view_leg_status","legsTable");
}

loadTables();
setInterval(loadTables,15000);
</script>
